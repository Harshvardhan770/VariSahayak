# VARI Sahayak Lost & Found face-matching service.

# adapted from. Do not substitute another face-recognition library: embeddings already

# from a different model. Changing MODEL_NAME means re-enrolling every profile.

Flask==3.0.3                       # HTTP layer
Flask-Cors==4.0.1                  # controlled cross-origin access for the backend
pymongo==4.10.1                    # the only datastore this service has
python-dotenv==1.0.1               # configuration
gunicorn==23.0.0                   # WSGI server in the container
